  test_subject <- subjects[i]

  test_data  <- vrmid1_pupil_svr_data %>% filter(Subject == test_subject) %>% drop_na()

  if (nrow(train_data) < 2) {
    warning(paste("Skipping subject", test_subject, "due to insufficient training data."))
    next
  }
  
  # Extract matrices
  x_train <- as.matrix(train_data[, feature_cols])
  y_train <- train_data$arousal
  
  x_test  <- as.matrix(test_data[, feature_cols])
  y_test  <- test_data$arousal
  
  # Fit LASSO CV model (alpha=1 for pure lasso)
  cv_model <- cv.glmnet(x_train, y_train, alpha = 1, standardize = TRUE)
  best_lambda <- cv_model$lambda.min
  
  # Final model
  model <- glmnet(x_train, y_train, alpha = 1, lambda = best_lambda, standardize = FALSE)
  
  # Predictions
  preds <- predict(model, newx = x_test)
  preds <- as.numeric(preds)
  
  # Identify rows with any NA
  row_all_na <- apply(cbind(x_test, y_test), 1, function(x) any(is.na(x)))
  
  preds_clean <- preds[!row_all_na]
  true_vals_clean <- y_test[!row_all_na]
  
  # Compute LOSO metrics
  train_mean <- mean(y_train, na.rm = TRUE)
  
  r2_cv <- 1 - sum((true_vals_clean - preds_clean)^2) /
    sum((true_vals_clean - train_mean)^2)
  
  rmse <- sqrt(mean((true_vals_clean - preds_clean)^2))

    # Find closest lambda in inner_model$lambda
    lambda_diff <- abs(inner_model$lambda - fused_model$lambda[l])
    closest_idx <- which.min(lambda_diff)
    
    beta_l <- inner_model$beta[, closest_idx]

# Store subject-wise results

feature_cols <- as.character(2:(time_end*120+1))
subjects <- unique(fmri3_pupil_svr_data$subject)
n_subjects <- length(subjects)

# Weight matrix (subject × features)
weights_fmri3_out_lasso <- matrix(NA, nrow = n_subjects, ncol = length(feature_cols))
colnames(weights_fmri3_out_lasso) <- feature_cols


for (i in seq_along(subjects)) {

  test_subject <- subjects[i]
  
  train_data <- fmri3_pupil_svr_data %>% filter(subject != test_subject) %>% drop_na()
  test_data  <- fmri3_pupil_svr_data %>% filter(subject == test_subject) %>% drop_na()
  
  # Skip if too little training data
  if (nrow(train_data) < 2) {
    warning(paste("Skipping subject", test_subject, "due to insufficient training data."))
    next
  }
  
  # Extract matrices
  x_train <- as.matrix(train_data[, feature_cols])
  y_train <- train_data$arousal
  
  x_test  <- as.matrix(test_data[, feature_cols])
  y_test  <- test_data$arousal
  
  # Fit LASSO CV model (alpha=1 for pure lasso)
  cv_model <- cv.glmnet(x_train, y_train, alpha = 1, standardize = TRUE)
  best_lambda <- cv_model$lambda.min
  
  # Final model
  model <- glmnet(x_train, y_train, alpha = 1, lambda = best_lambda, standardize = FALSE)
  
  # Predictions
  preds <- predict(model, newx = x_test)
  preds <- as.numeric(preds)
  
  # Identify rows with any NA
  row_all_na <- apply(cbind(x_test, y_test), 1, function(x) any(is.na(x)))
  
  preds_clean <- preds[!row_all_na]
  true_vals_clean <- y_test[!row_all_na]
  
  # Compute LOSO metrics
  train_mean <- mean(y_train, na.rm = TRUE)
  
  r2_cv <- 1 - sum((true_vals_clean - preds_clean)^2) /
    sum((true_vals_clean - train_mean)^2)
  
  rmse <- sqrt(mean((true_vals_clean - preds_clean)^2))
  
  acc <- cor(preds_clean, true_vals_clean, use = "complete.obs")
  
  # Store subject-wise results
  weights_fmri3_out_lasso_results <- rbind(
    weights_fmri3_out_lasso_results,
    data.frame(subject = test_subject, RMSE = rmse, R2 = r2_cv, acc = acc)
  )
  
  # Store pooled predictions
  weights_fmri3_out_lasso_all_preds <- rbind(
    weights_fmri3_out_lasso_all_preds,
    data.frame(subject = test_subject, y_true = true_vals_clean, y_pred = preds_clean)
  )
  
  # Feature weights
  weights_fmri3_out_lasso[i, ] <- as.numeric(model$beta)
}
