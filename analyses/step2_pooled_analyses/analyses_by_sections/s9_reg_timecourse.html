# threshold and min cluster length
p_thresh <- 1e-3
mark_clusters <- function(df, p_thresh, min_cluster_length, merge_gap) {
  # df must contain columns: p_val_fdr and sample_in_sec (or any per-row index)
  # min_cluster_length and merge_gap are in SAMPLES (not seconds).
  
  # Ensure ordering and create sample index (1..N)
  df <- df %>%
    arrange(Time_sec, sample_in_sec) %>%
    mutate(.row_idx = seq_len(n()))
  
  # Logical vector: below threshold
  below <- df$p_val_fdr < p_thresh
  
  # Identify runs of TRUE / FALSE using rle
  r <- rle(below)
  lens <- r$lengths
  vals <- r$values
  ends <- cumsum(lens)
  starts <- ends - lens + 1
  runs <- tibble(
    run_id = seq_along(lens),
    start_idx = starts,
    end_idx = ends,
    length = lens,
    is_true = vals
  )
  
  # Keep only TRUE runs (candidate significant clusters)
  true_runs <- runs %>% filter(is_true)
  
  if (nrow(true_runs) == 0) {
    df$sig_cluster <- NA
    return(df %>% select(-.row_idx))
  }
  
  # Compute gaps between consecutive TRUE runs
  true_runs <- true_runs %>%
    arrange(start_idx) %>%
    mutate(
      prev_end = lag(end_idx),
      gap = if_else(is.na(prev_end), NA, start_idx - prev_end - 1L)
    )
  
  # start new merge group whenever gap > merge_gap or NA
  true_runs <- true_runs %>%
    mutate(new_group = cumsum(if_else(is.na(gap) | gap > merge_gap, 1L, 0L)))
  
  # Summarize merged groups
  merged <- true_runs %>%
    group_by(new_group) %>%
    summarise(
      start_idx = first(start_idx),
      end_idx = last(end_idx),
      size = sum(length),
      .groups = "drop"
    ) %>%
    filter(size >= min_cluster_length)
  
  # If no merged cluster passes min length, return NA sig_cluster
  if (nrow(merged) == 0) {
    df$sig_cluster <- NA
    return(df %>% select(-.row_idx))
  }
  
  # Create logical mask of significant samples
  sig_mask <- rep(FALSE, nrow(df))
  for (i in seq_len(nrow(merged))) {
    si <- merged$start_idx[i]
    ei <- merged$end_idx[i]
    si <- max(1L, as.integer(si))
    ei <- min(nrow(df), as.integer(ei))
    sig_mask[si:ei] <- TRUE
  }
  
  # Assign sig_cluster column (1 = significant, NA = not)
  df$sig_cluster <- ifelse(sig_mask, 1L, NA)
  
  # Drop helper column and return
  df <- df %>% select(-.row_idx)
  return(df)
}
